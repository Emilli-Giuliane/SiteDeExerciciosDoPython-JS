Exercício 08

Faça um programa que peça a idade e a altura de 5 pessoas, armazene cada informação no seu respectivo vetor. Imprima a idade e a altura na ordem inversa a ordem lida. 

let idade = [];
let altura = [];

for (let i = 0; i < 5; i++) {
    idade.push(parseInt(prompt(`Digite a idade da ${i+1}º pessoa.`)));
    altura.push(parseFloat(prompt(`Digite a altura da ${i+1}º pessoa.`)));                
}
let saidaConsole = `===== Mostrando números digitados =====\n\nIdades: ${idade}.\nAlturas: ${altura}.\n\n===== Mostrando na ordem inversa de como são lidos =====\n\n`;
idade.sort((a,b) => a - b);
altura.sort((a,b) => a - b);

saidaConsole += `Idades: ${idade}.\nAlturas: ${altura}`;

function log(msg) {
    document.getElementById("console").innerHTML = msg;
};
log(saidaConsole);
