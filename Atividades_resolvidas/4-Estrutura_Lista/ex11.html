Exercício 11

Altere o programa anterior, intercalando 3 vetores de 10 elementos cada. 

let a = [];
let b = [];
let c = [];
let tudoJunto = [];
let cont = {
    "a": 0,
    "b": 1,
    "c": 2
}

for (let i = 0; i < 10; i++) {
        a.push(parseInt(prompt(`Digite o ${i + 1}º valor do primeiro vetor.`)));
        b.push(parseInt(prompt(`Digite o ${i + 1}º valor do segundo vetor.`)));
        c.push(parseInt(prompt(`Digite o ${i + 1}º valor do terceiro vetor.`)))
};
    
for (let i = 0; i < 10; i++) {
    tudoJunto[cont.a] = (a[i]);
    tudoJunto[cont.b] = (b[i]);
    tudoJunto[cont.c] = (c[i]);

    cont.a += 3;
    cont.b += 3;
    cont.c += 3;
};

let saidaConsole = `===== Valores de A =====\n${a}\n\n===== Valores de B =====\n${b}\n\n===== Valores de C =====\n${c}\n\n===== Todos os valores juntos =====\n${tudoJunto}`;
