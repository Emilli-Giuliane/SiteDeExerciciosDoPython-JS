Exercício 07

Faça um programa que leia um vetor de 5 números inteiros, mostre a soma, a multiplicação e os números. 

let num = [];
let soma = 0;
let mult = 1;

for (let i = 0; i < 5; i++) {
    num.push(parseInt(prompt(`Digite o ${i+1}º número.`)));
    soma += num[i];
    mult *= num[i];
}

let saidaConsole = `Números escolhidos: ${num}.\nA soma de todos os números: ${soma}.\nO resultado da multiplicação dos números é: ${mult}`;

function log(msg) {
    document.getElementById("console").innerHTML = msg;
};
log(saidaConsole);
