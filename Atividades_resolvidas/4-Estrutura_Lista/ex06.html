Exercício 06

Faça um programa que peça as quatro notas de 10 alunos, calcule e armazene num vetor a média de cada aluno, imprima o número de alunos com média maior ou igual a 7.0. 

let alunos = {};
let contAlunosNota7 = 0;

for (let i = 0; i < 10; i++) {
    alunos[i] = {}

    alunos[i].nome = prompt("Informe o nome do aluno " + (i + 1) + ".")
    
    alunos[i].notas = [];

    alunos[i].media = 0;
    
    for (let j = 0; j < 4; j++){
        alunos[i].notas.push(parseFloat(prompt(`Digite a ${j+1}º nota do aluno ${alunos[i].nome}`)))
        alunos[i].media += alunos[i].notas[j]
    }
    alunos[i].media /= 4;

    if (alunos[i].media >= 7.0) {
        contAlunosNota7++;
    }
}
let saidaConsole = `${contAlunosNota7} alunos estão com a média maior que 7.0.`;
console.log(alunos)
