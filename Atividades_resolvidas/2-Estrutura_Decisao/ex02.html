Exercício 02

Faça um programa que peça um valor e mostre na tela se o valor é positivo ou negativo. 

alert("Para mostrar essa atividade, siga as instruções.\nEste programa mostrará se o número digitado é negativo ou positivo");
let num = parseInt(prompt("Digite um número inteiro, seja negativo ou positivo."));

let saidaConsole = ""
if(num >= 0){
    saidaConsole = `O número ${num} é um número positivo.`;
}else{
    saidaConsole = `O número ${num} é um número negativo.`;
}

console.log(saidaConsole);
