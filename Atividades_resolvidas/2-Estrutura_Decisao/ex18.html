Exercício 18

Faça um programa que peça uma data no formato dd/mm/aaaa e determine se a mesma é uma data válida. 

alert("Este programa calcula se a data informada é real ou não.");

let data = prompt("Digite a data (formato em dd/mm/aaaa)");
data = data.split("/");
let dia = parseInt(data[0]);
let mes = parseInt(data[1]);
let ano = parseInt(data[2]);
let saidaConsole = "";
let listaMes31 = [1,3,5,7,8,10,12];
let listaMes30 = [4,6,9,11];

if (listaMes31.includes(mes)) {
    if (dia <= 31) {
        saidaConsole = `O dia ${dia} do mês ${mes} do ano ${ano} existe no calendário real.`;
    } else {
        saidaConsole = "Essa data não é valida. Encerrando o programa. . ."
    }
} else if (listaMes30.includes(mes)) {
    if (dia <= 30) {
        saidaConsole = `O dia ${dia} do mês ${mes} do ano ${ano} existe no calendário real.`;
    } else {
        saidaConsole = "Essa data não é valida. Encerrando o programa. . ."
    }
} else if (mes == 2) {
    if ((ano % 4 === 0 && ano % 100 !== 0) || ano % 400 === 0) {
        if (dia <= 29){
            saidaConsole = `O dia ${dia} do mês ${mes} do ano ${ano} existe no calendário real.`;
        } else {
            saidaConsole = "Essa data não é valida. Encerrando o programa. . ."
        }
    } else {
        if (dia <= 28) {
            saidaConsole = `O dia ${dia} do mês ${mes} do ano ${ano} existe no calendário real.`;
        } else {
            saidaConsole = "Essa data não é valida. Encerrando o programa. . ."
        }

} else {
