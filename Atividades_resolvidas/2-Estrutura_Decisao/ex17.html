Exercício 17

Faça um programa que peça um número correspondente a um determinado ano e em seguida informe se este ano é ou não bissexto. 

alert("Este programa calcula se um ano é bissexto ou não.");

let ano = parseInt(prompt("Digite o ano (Ex.:1900)"));
let saidaConsole = "";

if ((ano % 4 === 0 && ano % 100 !== 0) || ano % 400 === 0) {
    saidaConsole = "O ano " + ano + " é bissexto";
} else {
    saidaConsole = "O ano " + ano + " não é bissexto";
}

console.log(saidaConsole);
