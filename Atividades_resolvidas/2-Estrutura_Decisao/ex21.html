Exercício 21

Faça um programa para um caixa eletrônico. O programa deverá perguntar ao usuário a valor do saque e depois informar quantas notas de cada valor serão fornecidas. As notas disponíveis serão as de 1, 5, 10, 50 e 100 reais. O valor mínimo é de 10 reais e o máximo de 600 reais. O programa não deve se preocupar com a quantidade de notas existentes na máquina.

  • Exemplo 1: Para sacar a quantia de 256 reais, o programa fornece duas notas de 100, uma nota de 50, uma nota de 5 e uma nota de 1;
  • Exemplo 2: Para sacar a quantia de 399 reais, o programa fornece três notas de 100, uma nota de 50, quatro notas de 10, uma nota de 5 e quatro notas de 1.
alert("Siga as instruções para ver esse programa funcionando.\nEsse programa é uma simulação de caixa que irá ler uma quantia menor que R$600 e imprimir a quantidade de notas em dinheiro necessárias para chegar até o valor desejado.");
let numero = parseFloat(prompt("Digite uma quantia menor que R$600 e que seja maior que R$10."));
let numString = numero.toString().split("");
let notasDinheiro = [1,5,10,50,100];
let saidaConsole = `Valor digitado: R$${numero}`;

if (numero <= 600 && numero >= 100) {
    if (numero >= 100) {
        saidaConsole += `\n${numString[0]} notas de R$${notasDinheiro[4]}`;
    }
    if ((numString[1] * 10) >= notasDinheiro[3]) { // 50
        saidaConsole += `\nUma nota de R$${notasDinheiro[3]}`;
        
    } else if ((numString[1] * 10) >= notasDinheiro[2]) {
        saidaConsole += `\n${numString[1]} notas de R$${notasDinheiro[2]}`;
    }
    if (((numString[1] * 10) - notasDinheiro[3]) >= notasDinheiro[2]) { //10
        saidaConsole += `\n${((numString[1] * 10) - notasDinheiro[3]) / notasDinheiro[2]} notas de R$${notasDinheiro[2]}`;
    }
    if (numString[2] >= 5) {
        saidaConsole += `\nUma nota de R$${notasDinheiro[1]}`;
        if ((numString[2] - 5) < 5 && (numString[2] - 5) !== 0) {
            saidaConsole += `\n${numString[2] - notasDinheiro[1]} notas de R$${notasDinheiro[0]}`;
        }
    } 
    if ((numString[2]) < 5 && numString[2] != 0) {
        saidaConsole += `\n${numString[2] * 1} notas de R$${notasDinheiro[0]}`;
    }
    // TODO: Fazer o mesmo para as dezenas e unidades
} else if (numero >= 50 && numero < 100) {
    if ((numString[0] * 10) >= notasDinheiro[3]) { // 50
        saidaConsole += `\nUma nota de R$${notasDinheiro[3]}`;
    } else if ((numString[0] * 10) >= notasDinheiro[2]) {
        saidaConsole += `\n${numString[0]} notas de R$${notasDinheiro[2]}`;
    }
    if (((numString[0] * 10) - notasDinheiro[3]) >= notasDinheiro[2]) { //10
        saidaConsole += `\n${((numString[0] * 10) - notasDinheiro[3]) / notasDinheiro[2]} notas de R$${notasDinheiro[2]}`;
    }
    if (numString[1] >= 5) {
        saidaConsole += `\nUma nota de R$${notasDinheiro[1]}`;
        if ((numString[1] - 5) < 5 && (numString[1] - 5) !== 0) {
            saidaConsole += `\n${numString[1] - notasDinheiro[1]} notas de R$${notasDinheiro[0]}`;
        }
    } 
    if ((numString[1]) < 5 && numString[1] != 0) {
        saidaConsole += `\n${numString[1] * 1} notas de R$${notasDinheiro[0]}`;
    }
} else if (numero >= 10 && numero < 50) {
    saidaConsole += `\n${numString[0]} notas de R$${notasDinheiro[2]}`;
    if (numString[1] >= 5) {
        saidaConsole += `\nUma nota de ${notasDinheiro[1]}`;
        if ((numString[1] - 5) < 5 && (numString[1] - 5) !== 0) {
            saidaConsole += `\n${numString[1] - notasDinheiro[1]} notas de R$${notasDinheiro[0]}`;
        }
    } 
    if ((numString[1]) < 5 && numString[1] != 0) {
        saidaConsole += `\n${numString[1] * 1} notas de R$${notasDinheiro[0]}`;
    }
} else if (numero < 10) {
    saidaConsole = `[ERRO] Valor muito baixo. Por favor escolha um valor entre R$600 e R$10`;
} else if (numero > 600) {
    saidaConsole = `[ERRO] Valor muito alto. Por favor escolha um valor entre R$600 e R$10`;
}

console.log(saidaConsole);
