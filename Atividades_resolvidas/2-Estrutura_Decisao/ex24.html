Exercício 24

Faça um programa que leia 2 números e em seguida pergunte ao usuário qual operação ele deseja realizar. O resultado da operação deve ser acompanhado de uma frase que diga se o número é:

  • par ou ímpar;
  • positivo ou negativo;
  • inteiro ou decimal.
let num1 = parseFloat(prompt("Digite um número"));
let num2 = parseFloat(prompt("Digite mais um número"));
let resultado = 0;
let operacao = parseInt(prompt("Qual operação gostaria de fazer? Digite em números para esconlher uma das opções a baixo\n\n1 - Somar;\n2 - Subtrair;\n3 - Multiplicar;\n4 - Dividir."));
let saidaConsole = "";

switch (operacao) {
    case 1:
        resultado = num1 + num2;
        saidaConsole = `O resultado de ${num1} + ${num2} é: ${resultado}.`
        break;
    case 2:
        resultado = num1 - num2;
        saidaConsole = `O resultado de ${num1} - ${num2} é: ${resultado}.`
        break;
    case 3:
        resultado = num1 * num2;
        saidaConsole = `O resultado de ${num1} * ${num2} é: ${resultado}.`
        break;
    case 4:
        resultado = num1 / num2;
        saidaConsole = `O resultado de ${num1} / ${num2} é: ${resultado}.`
        break;
    default:
        saidaConsole = "Opção inválida. Por favor digite uma opção válida."
}
// Impar ou Par
if (resultado % 2 === 0) {
    saidaConsole += "\nO número " + resultado + " é um número PAR!";
} else {
    saidaConsole += "\nO número " + resultado + " é um número IMPAR!";
}

// Positivo ou Negativo
if (resultado >= 0){
    saidaConsole += `\nO número ${resultado} é um número POSITIVO`;
} else {
    saidaConsole += `\nO número ${resultado} é um número NEGATIVO`;
}

// Inteiro ou Decimal
if (((resultado.toFixed(2)).toString()).includes(".00")) {
    saidaConsole += `\nO número ${resultado} é um inteiro.`;
} else {
    saidaConsole += `\nO número ${resultado} é um decimal.`;
}


console.log(saidaConsole);
