Estrutura de decisão

Exercício 08

Faça um programa que pergunte o preço de três produtos e informe qual produto você deve comprar, sabendo que a decisão é sempre pelo mais barato: 

let valorLeite = parseFloat(prompt("Digite o valor do leite"));
let valorVerdura = parseFloat(prompt("Digite o valor da verdura"));
let valorBolo = parseFloat(prompt("Digite o valor do bolo"));

let saidaConsole = "";
if (valorLeite < valorVerdura && valorLeite < valorBolo) {
    saidaConsole = `O valor do leite está mais barato`;
} else if (valorVerdura < valorLeite && valorVerdura < valorBolo) {
    saidaConsole = `O valor da verdura está mais barato`;
} else if (valorBolo < valorLeite && valorBolo < valorVerdura) {
    saidaConsole = `O valor do bolo está mais barato`;
}

console.log(saidaConsole);
