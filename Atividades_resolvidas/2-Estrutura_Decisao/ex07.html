Exercício 07

Faça um programa que leia três números e mostre o maior e o menor deles: 

let n1 = parseInt(prompt("Digite o primeiro número"));
let n2 = parseInt(prompt("Digite o segundo número"));
let n3 = parseInt(prompt("Digite o terceiro número"));

let saidaConsole = "";
if (n1 > n2 && n1 > n3) {
    saidaConsole = `O número "${n1}" é o maior de todos.`;
    
    if (n2 > n3) {
        saidaConsole += `\nO número "${n3}" é menor de todos.`;
    } else if (n3 > n2) {
        saidaConsole += `\nO número "${n2}" é o menor de todos`;
    };
    
} else if (n2 > n1 && n2 > n3) {
    saidaConsole = `O número "${n2}" é o maior de todos.`;

    if (n1 > n3) {
        saidaConsole += `\nO número "${n3}" é o menor de todos`;
    } else if (n3 > n1) {
        saidaConsole += `\nO número "${n1}" é o menor de todos`;
    };
    
} else if (n3 > n1 && n3 > n2) {
    saidaConsole = `O número "${n3}" é o maior de todos.`;

    if (n1 > n2) {
        saidaConsole += `\nO número "${n2}" é o menor de todos`;
    } else if (n2 > n1) {
        saidaConsole += `\nO número "${n1}" é o menor de todos`;
    };

} else {
    saidaConsole = "São números todos iguais";
};

console.log(saidaConsole);
