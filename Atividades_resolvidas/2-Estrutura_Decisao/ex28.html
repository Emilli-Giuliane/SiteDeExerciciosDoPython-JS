Exercício 28

O Hipermercado Tabajara está com uma promoção de carnes que é imperdível. Confira: 

                     Até 5 Kg                   Acima de 5 Kg
File Duplo      R$ 4,90 por Kg          R$ 5,80 por Kg
Alcatra         R$ 5,90 por Kg          R$ 6,80 por Kg
Picanha         R$ 6,90 por Kg          R$ 7,80 por Kg

Para atender a todos os clientes, cada cliente poderá levar apenas um dos tipos de carne da promoção, porém não há limites para a quantidade de carne por cliente. Se compra for feita no cartão Tabajara o cliente receberá ainda um desconto de 5% sobre o total da compra. Escreva um programa que peça o tipo e a quantidade de carne comprada pelo usuário e gere um cupom fiscal, contendo as informações da compra: tipo e quantidade de carne, preço total, tipo de pagamento, valor do desconto e valor a pagar. 

alert("Por favor, responda as perguntas a seguir com \"s\" para \"Sim\" ou \"n\" para \"Não\".");

let tipoCarne = prompt("Digite o número correspondente a carne que você deseja comprar:\n1 - Filé Duplo;\n2 - Alcatra;\n3 - Picanha.");

let qtdFile = 0;
let qtdAlcatra = 0;
let qtdPicanha = 0;

let totalFile = 0;
let totalAlcatra = 0;
let totalPicanha = 0;
let totalValor = 0;
let saidaConsole = `\nValor a ser pago por:`;

if (tipoCarne === "1") {
    qtdFile = parseFloat(prompt("Digite a quantidade em Kg de filé Duplo a ser comprado."));

    if (qtdFile <= 5) {
        totalFile = 4.90 * qtdFile;
    } else {
        totalFile = 5.80 * qtdFile;
    }
    saidaConsole += `\nFile = R$${totalFile.toFixed(2)}.`;

}
if (tipoCarne === "2") {
    qtdAlcatra = parseFloat(prompt("Digite a quantidade em Kg de Alcatra a ser comprado."));

    if (qtdAlcatra <= 5) {
        totalAlcatra = 5.90 * qtdAlcatra;
    } else {
        totalAlcatra = 6.80 * qtdAlcatra;
    }
    saidaConsole += `\nAlcatra = R$${totalAlcatra.toFixed(2)}.`;
}
if (tipoCarne === "3") {
    qtdPicanha = parseFloat(prompt("Digite a quantidade em Kg de Picanha a ser comprado."));

    if (qtdPicanha <= 5) {
        totalPicanha = 6.90 * qtdPicanha;
    } else {
        totalPicanha = 7.80 * qtdPicanha;
    }
    saidaConsole += `\nPicanha = R$${totalPicanha.toFixed(2)}.\nValor Total = R$${totalValor.toFixed(2)}.`;
}

// saidaConsole += `\nValor a ser pago por:\nFile = R$${totalFile.toFixed(2)}.\nAlcatra = R$${totalAlcatra.toFixed(2)}.\nPicanha = R$${totalPicanha.toFixed(2)}.\nValor Total = R$${totalValor.toFixed(2)}.`;

totalValor = totalFile + totalAlcatra + totalPicanha;

let possuiCartao = prompt("Você possui o cartão da loja?\ns - Sim;\nn - Não").toLowerCase();

if (possuiCartao === "s") {
    totalValor -= (totalValor * 0.05);

    saidaConsole += `\nVocê ganhou desconto de 5%. O novo valor a ser pago será de R$${totalValor.toFixed(2)}`;
}

console.log(saidaConsole);
function log(msg) {
    document.getElementById("console").innerHTML += msg + "\n";
};
log(saidaConsole);
